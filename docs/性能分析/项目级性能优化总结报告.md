**当前问题**：每个页面各自实现缓存逻辑，代码重复

```typescript
// src/utils/cache.ts
class CacheManager<T> {
  private cache = new Map<string, T>();
  private maxSize: number;

  constructor(maxSize = 50) {
    this.maxSize = maxSize;
  }

  get(key: string): T | undefined {
    return this.cache.get(key);
  }

  set(key: string, value: T): void {
    if (this.cache.size >= this.maxSize) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }
    this.cache.set(key, value);
  }

  clear(): void {
    this.cache.clear();
  }
}

// 全局实例
export const chapterCache = new CacheManager<FlashCard[]>(10);
export const jobCache = new CacheManager<CampusJobData[]>(5);
export const pdfTextCache = new CacheManager<string>(20);
```

**使用示例**：
```typescript
// 替代各页面的 Map 缓存
```

#### 优化 2: 统一 Worker 管理

**当前问题**：多个页面可能需要 Worker，各自管理生命周期

```typescript
// src/workers/WorkerPool.ts
class WorkerPool {
  private workers: Map<string, Worker> = new Map();

  getWorker(name: string, scriptUrl: string): Worker {
    if (!this.workers.has(name)) {
      const worker = new Worker(scriptUrl, { type: 'module' });
      this.workers.set(name, worker);
    }
    return this.workers.get(name)!;
  }

  terminateWorker(name: string): void {
    const worker = this.workers.get(name);
    if (worker) {
      worker.terminate();
      this.workers.delete(name);
    }
  }

  terminateAll(): void {
    this.workers.forEach(worker => worker.terminate());
    this.workers.clear();
  }
}

export const workerPool = new WorkerPool();

// 使用
const pdfWorker = workerPool.getWorker('pdf-extract', '/pdf-extract.worker.js');
pdfWorker.postMessage({ dataUrl });
```

**收益**：
- Worker 复用

```typescript
// Hook 封装
export function useProgressTask<T>(
  task: () => Promise<T>,
  message: string
): [execute: () => Promise<T>, progress: number, isLoading: boolean] {
  const [progress, setProgress] = useState(0);
  const [isLoading, setIsLoading] = useState(false);

  const execute = async () => {
    setIsLoading(true);
    setProgress(0);
    
    const timer = setInterval(() => {
      setProgress(prev => Math.min(prev + 5, 90));
    }, 300);
    
    try {
      const result = await task();
      clearInterval(timer);
      setProgress(100);
      return result;
    } finally {
      setTimeout(() => {
        setIsLoading(false);
        setProgress(0);
      }, 500);
    }
  };

  return [execute, progress, isLoading];
}
```

```typescript
interface PerformanceMetric {

class PerformanceMonitor {

    this.recordMetric(name, duration);
    return result;
  }
  
  // 记录异步函数耗时
  async measureAsync<T>(name: string, fn: () => Promise<T>): Promise<T> {
    const start = performance.now();
    const result = await fn();
    const duration = performance.now() - start;
    
    this.recordMetric(name, duration);
    return result;
  }
  
  // 记录指标
  recordMetric(name: string, value: number): void {
    this.metrics.push({
      name,
      value,
      timestamp: Date.now(),
      page: window.location.pathname,
    });
    
    // 上报到分析平台（可选）
    if (this.metrics.length >= 10) {
      this.flush();
    }
  }
  
  // 上报指标
  private flush(): void {
    console.log('[Performance Metrics]', this.metrics);
    // TODO: 发送到分析后端
    this.metrics = [];
  }
  
  // 获取性能报告
  getReport(): Record<string, { avg: number; max: number; min: number; count: number }> {
    const report: Record<string, number[]> = {};
    
    this.metrics.forEach(({ name, value }) => {
      if (!report[name]) report[name] = [];
      report[name].push(value);
    });
    
    return Object.fromEntries(
      Object.entries(report).map(([name, values]) => [
        name,
        {
          avg: values.reduce((a, b) => a + b, 0) / values.length,
          max: Math.max(...values),
          min: Math.min(...values),
          count: values.length,
        },
      ])
    );
  }
}

    // 原有逻辑
```

1. **增量计数器**：在 Zustand store 中维护 `totalRemembered` 字段
2. **增量更新**：每次状态变化时，只更新计数器（+1 或 -1）
3. **初始化计算**：应用启动时计算一次，后续只做增量

状态更新响应从 250ms 降到 5ms（↓98%）。

同时我还做了：
- **选择器优化**：使用精确订阅 + shallow 比较，减少不必要的重渲染
- **Store 拆分**：按功能域拆分（数据 store + 状态 store + UI store），减少更新影响范围

#### Q4: 如何处理云端同步与用户体验的平衡？

**回答**：
原本做法是页面加载时立即同步，弱网环境下用户需要等待 1.5 秒才能操作。

我改为**乐观更新 + 后台同步**：
1. 用户操作立即更新本地状态（Zustand），UI 立即响应
2. 后台防抖 1.5 秒后批量同步到 Supabase
3. 同步完成后，如果有云端冲突，显示 Toast 提示用户
4. 页面加载时也是先显示本地数据，2 秒后后台同步

首屏可交互时间从 1.5 秒降到 150ms（↓90%），用户无感知等待。
